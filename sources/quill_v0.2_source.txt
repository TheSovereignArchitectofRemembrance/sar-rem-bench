  function fk(q,cfg){
    const [l1,l2,l3]=cfg.lengths, a1=q[0],a2=q[0]+q[1],a3=a2+q[2],b=state.base;
    const p1={x:b.x+l1*Math.cos(a1),y:b.y+l1*Math.sin(a1)};
    const p2={x:p1.x+l2*Math.cos(a2),y:p1.y+l2*Math.sin(a2)};
    const p3={x:p2.x+l3*Math.cos(a3),y:p2.y+l3*Math.sin(a3)};
    return {base:{...b},p1,p2,p3,psi:wrap(a3)};
  }

  function solve3(A,b){
    const m=A.map((row,i)=>[...row,b[i]]);
    for(let col=0;col<3;col++){
      let pivot=col; for(let r=col+1;r<3;r++) if(Math.abs(m[r][col])>Math.abs(m[pivot][col])) pivot=r;
      if(Math.abs(m[pivot][col])<1e-10) return [0,0,0];
      [m[col],m[pivot]]=[m[pivot],m[col]];
      const d=m[col][col]; for(let j=col;j<4;j++)m[col][j]/=d;
      for(let r=0;r<3;r++)if(r!==col){const f=m[r][col];for(let j=col;j<4;j++)m[r][j]-=f*m[col][j];}
    }
    return [m[0][3],m[1][3],m[2][3]];
  }

  function nearestEquivalent(a,ref){ return a+Math.round((ref-a)/(2*Math.PI))*2*Math.PI; }
  function admissibleEquivalent(a,ref,lo,hi){
    const legal=[];for(let n=-2;n<=2;n++){const v=a+n*2*Math.PI;if(v>=lo-1e-9&&v<=hi+1e-9)legal.push(v);}
    if(legal.length)return legal.sort((u,v)=>Math.abs(u-ref)-Math.abs(v-ref))[0];
    return clamp(nearestEquivalent(a,ref),lo,hi);
  }
  function clampQ(q,cfg){ return q.map((v,i)=>clamp(v,cfg.mins[i],cfg.maxs[i])); }
  function objective(q,t,cfg){ const p=fk(q,cfg); const pe=Math.hypot(t.x-p.p3.x,t.y-p.p3.y), ae=Math.abs(wrap(t.psi-p.psi)); return pe*pe+(cfg.orientationWeight*ae)**2; }

  function analyticSeeds(t,prev,cfg){
    const [l1,l2,l3]=cfg.lengths, wx=t.x-l3*Math.cos(t.psi), wy=t.y-l3*Math.sin(t.psi);
    const dx=wx-state.base.x,dy=wy-state.base.y,r=Math.hypot(dx,dy);
    const raw=(r*r-l1*l1-l2*l2)/(2*l1*l2),cc=clamp(raw,-1,1), seeds=[];
    for(const sign of [1,-1]){
      const q2=sign*Math.acos(cc), q1=Math.atan2(dy,dx)-Math.atan2(l2*Math.sin(q2),l1+l2*Math.cos(q2));
      const base=[q1,q2,t.psi-q1-q2].map((v,i)=>admissibleEquivalent(v,prev[i],cfg.mins[i],cfg.maxs[i]));
      seeds.push({q:clampQ(base,cfg),branch:sign>0?'elbow +':'elbow −'});
    }
    return {seeds,wristRadius:r,geometricReach:raw>=-1-1e-9&&raw<=1+1e-9};
  }

  function refine(seed,t,cfg){
    let q=[...seed], activeHits=new Set();
    for(let iter=0;iter<36;iter++){
      const pose=fk(q,cfg), e=[t.x-pose.p3.x,t.y-pose.p3.y,cfg.orientationWeight*wrap(t.psi-pose.psi)];
      if(Math.hypot(e[0],e[1])<.01&&Math.abs(e[2])<.002) break;
      const [l1,l2,l3]=cfg.lengths,a1=q[0],a2=q[0]+q[1],a3=a2+q[2],w=cfg.orientationWeight;
      const J=[
        [-l1*Math.sin(a1)-l2*Math.sin(a2)-l3*Math.sin(a3),-l2*Math.sin(a2)-l3*Math.sin(a3),-l3*Math.sin(a3)],
        [ l1*Math.cos(a1)+l2*Math.cos(a2)+l3*Math.cos(a3), l2*Math.cos(a2)+l3*Math.cos(a3), l3*Math.cos(a3)],
        [w,w,w]
      ];
      const lambda=3.5, A=Array.from({length:3},(_,r)=>Array.from({length:3},(_,c)=>J[r][0]*J[c][0]+J[r][1]*J[c][1]+J[r][2]*J[c][2]+(r===c?lambda*lambda:0)));
      const z=solve3(A,e), dq=[0,1,2].map(c=>J[0][c]*z[0]+J[1][c]*z[1]+J[2][c]*z[2]);
      for(let i=0;i<3;i++){
        q[i]+=clamp(dq[i],-.28,.28);
        const before=q[i]; q[i]=clamp(q[i],cfg.mins[i],cfg.maxs[i]); if(q[i]!==before)activeHits.add(i);
      }
    }
    return {q,activeHits};
  }

  function evaluatePose(q,t,cfg,branch){
    const pose=fk(q,cfg),positionError=hypot(pose.p3,t),angleError=Math.abs(wrap(pose.psi-t.psi));
    const active=[]; for(let i=0;i<3;i++) if(Math.min(Math.abs(q[i]-cfg.mins[i]),Math.abs(q[i]-cfg.maxs[i]))<rad(.45))active.push(`q${i+1}`);
    return {q,pose,positionError,angleError,pass:positionError<=cfg.positionTolerance&&angleError<=cfg.angleTolerance,active,branch:branch||(q[1]>=0?'elbow +':'elbow −'),singularity:Math.abs(Math.sin(q[1]))<.025};
  }

  function solvePose(t,prev,cfg){
    const analytic=analyticSeeds(t,prev,cfg),candidates=[];
    for(const seed of analytic.seeds){const refined=refine(seed.q,t,cfg);candidates.push(evaluatePose(refined.q,t,cfg,seed.branch));}
    const carry=refine(clampQ([...prev],cfg),t,cfg);candidates.push(evaluatePose(carry.q,t,cfg));
    let best=null;
    for(const candidate of candidates){const score=objective(candidate.q,t,cfg)+.015*candidate.q.reduce((s,v,i)=>s+(v-prev[i])**2,0);if(!best||score<best.score)best={...candidate,score};}
    const recovery=[];
    for(const candidate of candidates.slice(0,2))if(candidate.pass&&!recovery.some(r=>r.branch===candidate.branch))recovery.push(candidate);
    best.recovery=recovery;best.branchCount=recovery.length;best.alternate=recovery.find(r=>r.branch!==best.branch)||null;
    const {positionError,angleError,active,singularity}=best;
    const pass=positionError<=cfg.positionTolerance&&angleError<=cfg.angleTolerance;
    let cause='pass';
    if(!pass){
      const [l1,l2]=cfg.lengths;
      if(analytic.wristRadius>l1+l2+.01)cause='outer pose reach';
      else if(analytic.wristRadius<Math.abs(l1-l2)-.01)cause='inner pose reach';
      else if(active.length)cause='joint limit';
      else if(singularity)cause='singularity';
      else cause='coupled pose mismatch';
    }
    return {...best,pass,cause,wristRadius:analytic.wristRadius,geometricReach:analytic.geometricReach};
  }

  function pointSeeds(t,prev,cfg){
    const [l1,l2,l3]=cfg.lengths,q3=clamp(0,cfg.mins[2],cfg.maxs[2]),l23=Math.hypot(l2+l3*Math.cos(q3),l3*Math.sin(q3)),beta=Math.atan2(l3*Math.sin(q3),l2+l3*Math.cos(q3)),dx=t.x-state.base.x,dy=t.y-state.base.y,r=Math.hypot(dx,dy);
    const raw=(r*r-l1*l1-l23*l23)/(2*l1*l23),cc=clamp(raw,-1,1),seeds=[];
    for(const sign of [1,-1]){
      const q2eff=sign*Math.acos(cc),q2=q2eff-beta,q1=Math.atan2(dy,dx)-Math.atan2(l23*Math.sin(q2eff),l1+l23*Math.cos(q2eff));
      seeds.push({q:clampQ([admissibleEquivalent(q1,prev[0],cfg.mins[0],cfg.maxs[0]),admissibleEquivalent(q2,prev[1],cfg.mins[1],cfg.maxs[1]),q3],cfg),branch:sign>0?'elbow +':'elbow −'});
    }
    return {seeds,radius:r,outer:l1+l23,inner:Math.abs(l1-l23),beta};
  }

  function refinePoint(seed,t,cfg){
    let q=[seed[0],seed[1],clamp(0,cfg.mins[2],cfg.maxs[2])];
    for(let iter=0;iter<32;iter++){
      const p=fk(q,cfg),ex=t.x-p.p3.x,ey=t.y-p.p3.y;if(Math.hypot(ex,ey)<.01)break;
      const [l1,l2,l3]=cfg.lengths,a1=q[0],a2=q[0]+q[1],a3=a2+q[2];
      const j00=-l1*Math.sin(a1)-l2*Math.sin(a2)-l3*Math.sin(a3),j01=-l2*Math.sin(a2)-l3*Math.sin(a3);
      const j10= l1*Math.cos(a1)+l2*Math.cos(a2)+l3*Math.cos(a3),j11= l2*Math.cos(a2)+l3*Math.cos(a3);
      const lambda=2.5,a=j00*j00+j10*j10+lambda*lambda,b=j00*j01+j10*j11,d=j01*j01+j11*j11+lambda*lambda;
      const r0=j00*ex+j10*ey,r1=j01*ex+j11*ey,det=a*d-b*b;
      if(Math.abs(det)<1e-9)break;
      q[0]=clamp(q[0]+clamp((d*r0-b*r1)/det,-.3,.3),cfg.mins[0],cfg.maxs[0]);
      q[1]=clamp(q[1]+clamp((a*r1-b*r0)/det,-.3,.3),cfg.mins[1],cfg.maxs[1]);
    }
    return q;
  }

  function solvePoint(t,prev,cfg){
    const analytic=pointSeeds(t,prev,cfg),candidates=analytic.seeds.map(seed=>evaluatePose(refinePoint(seed.q,t,cfg),t,cfg,seed.branch));
    const carry=evaluatePose(refinePoint(prev,t,cfg),t,cfg),all=[...candidates,carry];let best=null;
    for(const candidate of all){const score=candidate.positionError**2+.015*candidate.q.reduce((s,v,i)=>s+(v-prev[i])**2,0);if(!best||score<best.score)best={...candidate,score};}
    for(const candidate of all)candidate.pass=candidate.positionError<=cfg.positionTolerance;
    const recovery=[];for(const candidate of candidates)if(candidate.pass&&!recovery.some(r=>r.branch===candidate.branch))recovery.push(candidate);
    best.pass=best.positionError<=cfg.positionTolerance;best.recovery=recovery;best.branchCount=recovery.length;best.alternate=recovery.find(r=>r.branch!==best.branch)||null;best.singularity=Math.abs(Math.sin(best.q[1]+analytic.beta))<.025;
    let cause='pass';if(!best.pass){if(analytic.radius>analytic.outer+.01)cause='outer point reach';else if(analytic.radius<analytic.inner-.01)cause='inner point reach';else if(best.active.length)cause='joint limit';else if(best.singularity)cause='singularity';else cause='point mismatch';}
    return {...best,cause,wristRadius:analytic.radius,geometricReach:analytic.radius>=analytic.inner-.01&&analytic.radius<=analytic.outer+.01};
  }

  function buildLedger(samples){
    const rows=[]; let run=null;
    for(let i=0;i<samples.length;i++){
      const s=samples[i];
      if(s.pass){ if(run){rows.push(run);run=null;} continue; }
      const key=`${s.cause}|${s.active.join(',')}`;
      if(!run||run.key!==key){ if(run)rows.push(run); run={key,start:i,end:i,cause:s.cause,count:1,maxPosition:s.positionError,maxAngle:s.angleError,active:s.active.join(', ')||'—'}; }
      else { run.end=i;run.count++;run.maxPosition=Math.max(run.maxPosition,s.positionError);run.maxAngle=Math.max(run.maxAngle,s.angleError); }
    }
    if(run)rows.push(run); return rows;
  }

  function summarize(samples,cfg){
    const passed=samples.filter(s=>s.pass).length;
    const meanPosition=samples.reduce((a,s)=>a+s.positionError,0)/Math.max(1,samples.length);
    const meanAngle=samples.reduce((a,s)=>a+s.angleError,0)/Math.max(1,samples.length);
    const maxPosition=Math.max(0,...samples.map(s=>s.positionError));
    const targetLength=pathLength(state.target),actualLength=pathLength(samples.map(s=>s.pose.p3));
    const twoBranch=samples.filter(s=>s.branchCount===2).length;let branchSwitches=0,last=null;
    for(const s of samples)if(s.pass){if(last&&last!==s.branch)branchSwitches++;last=s.branch;}
    return {config:cfg,samples,ledger:buildLedger(samples),metrics:{passed,total:samples.length,reachableFraction:passed/Math.max(1,samples.length),meanPosition,maxPosition,meanAngle,targetLength,actualLength,lengthRatio:actualLength/Math.max(1e-9,targetLength),twoBranch,twoBranchFraction:twoBranch/Math.max(1,samples.length),branchSwitches}};
  }
